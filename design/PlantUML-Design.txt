05 - Activity Diagram - Module 1
@startuml
title Activity - Search Services/Parts Information
start
partition "Process at MainUI2.jsp" {
 :display the UI;
}
:click Search Services/Parts;
partition "Process at SearchServicesPartsUI2.jsp" {
 :display the UI;
 if (Search type?) then (Service)
  :select Service, enter keyword, click Search;
  partition "ServiceDAO2" {
   :getServicesByName();
  }
  :Get Service List;
 else (Spare Part)
  :select SparePart, enter keyword, click Search;
  partition "SparePartDAO2" {
   :getSparePartsByName();
  }
  :get Spare Part List;
 endif
 :got the data;
 :display the UI;
 :click on one Service/Part;
}
partition "Process at ServiceDetailsUI2.jsp / SparePartDetailsUI2.jsp" {
 :Receive the service/spare part Information;
 :Display the UI;
}
stop
@enduml
